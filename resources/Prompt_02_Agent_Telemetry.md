## STEP 3: OTel Tracer Setup + `demo.*` attribute helpers

Create `agents/common/otel_setup.py`. This module is the single place every agent imports for tracer configuration and for the mandatory `demo.*` boilerplate per PCD §10.

```python
"""
OpenTelemetry setup for all agents.

Standard opentelemetry-sdk + OTLP gRPC exporter pointing at the ADOT Collector
(port 4317 per PCD §10). Idempotent: calling setup_tracer() multiple times in
the same process is safe.
"""
from __future__ import annotations
import os
from contextlib import contextmanager
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.sdk.resources import Resource, SERVICE_NAME, SERVICE_VERSION
# Post-1.20 constant. The older name DEPLOYMENT_ENVIRONMENT was renamed.
from opentelemetry.sdk.resources import DEPLOYMENT_ENVIRONMENT_NAME
from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter

_TRACER_PROVIDER: TracerProvider | None = None


def setup_tracer(agent_name: str, agent_version: str = "1.0.0"):
    """Configure the global tracer. Safe to call multiple times."""
    global _TRACER_PROVIDER
    if _TRACER_PROVIDER is not None:
        return trace.get_tracer(f"agentcore-demo-test1-{agent_name}")

    endpoint = os.environ.get("OTEL_EXPORTER_OTLP_ENDPOINT", "http://adot-collector:4317")
    environment = os.environ.get("DEPLOYMENT_ENVIRONMENT", "dev")

    resource = Resource.create({
        SERVICE_NAME: f"agentcore-demo-test1-{agent_name}",
        SERVICE_VERSION: agent_version,
        DEPLOYMENT_ENVIRONMENT_NAME: environment,
    })
    provider = TracerProvider(resource=resource)
    provider.add_span_processor(BatchSpanProcessor(OTLPSpanExporter(endpoint=endpoint, insecure=True)))
    trace.set_tracer_provider(provider)
    _TRACER_PROVIDER = provider
    return trace.get_tracer(f"agentcore-demo-test1-{agent_name}")


def apply_demo_attributes(span, request: dict) -> None:
    """Attach mandatory demo.* attributes from an inbound request payload.

    Per PCD §10, every span representing an agent step MUST carry these.
    Cardinality discipline: HIGH-cardinality fields (trace_id, workflow_id, ...)
    land at the EMF root; the ADOT Collector's attributes/demo processor enforces this.
    """
    span.set_attribute("demo.trace_id", request.get("trace_id", "Unavailable"))
    span.set_attribute("demo.workflow_template_id", request.get("workflow_template_id", "Unavailable"))
    span.set_attribute("demo.workflow_id", request.get("workflow_id", "Unavailable"))
    span.set_attribute("demo.agent_run_id", request.get("agent_run_id", "Unavailable"))
    span.set_attribute("demo.step_id", request.get("step_id", "Unavailable"))
    span.set_attribute("demo.actor_id", request.get("actor_id", "Unavailable"))
    span.set_attribute("demo.data_classification",
                       request.get("execution_context", {}).get("data_classification", "INTERNAL"))
    span.set_attribute("demo.error_category", "none")
    # Low-cardinality routing fields (safe in Dimensions).
    span.set_attribute("demo.environment",
                       request.get("execution_context", {}).get("environment", "dev"))
    span.set_attribute("demo.task_type",
                       request.get("task", {}).get("task_type", "Unavailable"))
    span.set_attribute("demo.tier", "first_party")


@contextmanager
def agent_execution_span(tracer, request: dict, span_name: str = "AgentExecution"):
    """Context manager for the top-level agent span with mandatory attributes."""
    with tracer.start_as_current_span(span_name) as span:
        apply_demo_attributes(span, request)
        try:
            yield span
        except Exception as exc:
            span.set_attribute("demo.error_category", "system_failure")
            span.record_exception(exc)
            raise
```

## STEP 3b: OTel Meter Setup — numeric metric instruments

Spans alone do NOT produce CloudWatch metrics. To get the `AgentLatencyMs`, `TokensInput`, `TokensOutput`, `AgentCostUsd`, `WorkflowOutcome` metrics that Phase 7 dashboard reads, we register OpenTelemetry **metric instruments** in the same module. The same ADOT Collector receives both spans (traces pipeline) and metric measurements (metrics pipeline) and routes them via separate exporters.

Add this to `agents/common/otel_setup.py`:

```python
# --- Metric instruments (PCD §10, Deliverable_7 §5.3) ---------------------
from opentelemetry import metrics
from opentelemetry.sdk.metrics import MeterProvider
from opentelemetry.sdk.metrics.export import PeriodicExportingMetricReader
from opentelemetry.exporter.otlp.proto.grpc.metric_exporter import OTLPMetricExporter

_METER_PROVIDER: MeterProvider | None = None


def setup_meter(agent_name: str, agent_version: str = "1.0.0"):
    """Idempotent meter setup. Safe to call multiple times."""
    global _METER_PROVIDER
    if _METER_PROVIDER is not None:
        return metrics.get_meter(f"agentcore-demo-test1-{agent_name}")

    endpoint = os.environ.get("OTEL_EXPORTER_OTLP_ENDPOINT", "http://adot-collector:4317")
    resource = Resource.create({
        SERVICE_NAME: f"agentcore-demo-test1-{agent_name}",
        SERVICE_VERSION: agent_version,
        DEPLOYMENT_ENVIRONMENT_NAME: os.environ.get("DEPLOYMENT_ENVIRONMENT", "dev"),
    })
    exporter = OTLPMetricExporter(endpoint=endpoint, insecure=True)
    reader = PeriodicExportingMetricReader(exporter, export_interval_millis=10_000)  # 10 s
    provider = MeterProvider(resource=resource, metric_readers=[reader])
    metrics.set_meter_provider(provider)
    _METER_PROVIDER = provider
    return metrics.get_meter(f"agentcore-demo-test1-{agent_name}")


# Module-level instruments registered once at import time so agents can
# import them directly. Names are the canonical CloudWatch metric names
# per Deliverable_7 §5.3.
_meter = setup_meter("shared")
AGENT_LATENCY_MS = _meter.create_histogram("AgentLatencyMs", unit="ms",
                                            description="Per-invocation agent active time")
TOKENS_INPUT = _meter.create_counter("TokensInput", unit="1",
                                      description="Bedrock input tokens consumed")
TOKENS_OUTPUT = _meter.create_counter("TokensOutput", unit="1",
                                       description="Bedrock output tokens consumed")
AGENT_COST_USD = _meter.create_histogram("AgentCostUsd", unit="USD",
                                          description="Per-invocation USD cost")
WORKFLOW_OUTCOME = _meter.create_counter("WorkflowOutcome", unit="1",
                                          description="+1 per terminal workflow outcome")


def record_invocation_metrics(*, agent_id: str, task_type: str,
                               workflow_template_id: str,
                               environment: str,
                               latency_ms: float,
                               tokens_in: int, tokens_out: int,
                               cost_usd: float) -> None:
    """Record the standard per-invocation metrics. Call once per /invoke after
    the 8-block payload is built (PCD §10 cardinality discipline: only
    LOW-cardinality attributes go on metric data points)."""
    attrs = {
        "agent_id": agent_id,
        "task_type": task_type,
        "tier": "first_party",
        "environment": environment,
        "workflow_template_id": workflow_template_id,
    }
    AGENT_LATENCY_MS.record(latency_ms, attributes=attrs)
    TOKENS_INPUT.add(tokens_in, attributes=attrs)
    TOKENS_OUTPUT.add(tokens_out, attributes=attrs)
    AGENT_COST_USD.record(cost_usd, attributes=attrs)
```

## STEP 4: 8-Block Payload Builder (unified — used by Steps 5/6/7)

> **Important (May 2026 correction).** An earlier draft of this prompt defined `build_complete_payload(status_code, trace_id, workflow_id, ...)` with a different signature than what Steps 5/6/7 actually call. The two were incompatible. This rewrite uses the **unified signature** that the agent code in Steps 5/6/7 expects: pass the full inbound `request` dict (for identifier echo) plus optional per-block overrides. The `waiting_for_human_ms` field can be set via the `waiting_for_human_ms` kwarg (default 0 — see PCD §6 and Guideline §8.4: the agent's own wait is 0; the workflow aggregates HITL wait time across steps).

Create `agents/common/payload_builder.py`:

```python
"""
8-Block Payload Builder — unified, raw dict, no Pydantic.

Reference: PCD §6, PAYLOAD_SCHEMA §3/§4, Guideline §8.

Single entry point `build_complete_payload(request, agent_id, ...)`. Reads the
inbound request for identifier echo (Block 1). Accepts per-block overrides so
the agent code can supply specific values (cost model, artifacts, tool calls,
risk findings). Validates the result against PAYLOAD_SCHEMA §9.

Cooperation with HITL:
  - `waiting_for_human_ms` defaults to 0 on the agent side. Per Guideline §8.4,
    the agent does NOT wait — it emits `requires_human_review` and exits. The
    workflow durably waits and tracks wait time at the workflow level. For
    debugging / demo visibility you MAY override `waiting_for_human_ms` to a
    non-zero value when emitting a HITL question; this is purely cosmetic and
    does not affect workflow accounting.
"""
from __future__ import annotations
import hashlib
import time
from typing import Any


UNAVAILABLE = "Unavailable"
REQUIRED_BLOCKS = {"status", "resources", "timing", "financial",
                    "artifacts", "quality", "tool_calls", "risk"}


def _sha256(content: bytes | str) -> str:
    if not content or content == UNAVAILABLE:
        return UNAVAILABLE
    if isinstance(content, str):
        content = content.encode()
    return "sha256:" + hashlib.sha256(content).hexdigest()


def _resources_for(cost_reporting_model: str, *, token_based=None,
                    vendor_cost=None, fixed_subscription=None) -> dict:
    """Block 2 — Resources."""
    block = {"cost_reporting_model": cost_reporting_model}
    if cost_reporting_model in {"token_based", "both"} and token_based:
        block["token_based"] = token_based
    if cost_reporting_model in {"vendor_cost", "both"} and vendor_cost:
        block["vendor_cost"] = vendor_cost
    if cost_reporting_model == "fixed_subscription" and fixed_subscription:
        block["fixed_subscription"] = fixed_subscription
    return block


def build_complete_payload(
    *,
    # Identifier echo source (REQUIRED)
    request: dict,
    agent_id: str,
    agent_version: str,
    # Timing
    started_at_ms: int,
    ended_at_ms: int,
    waiting_for_human_ms: int = 0,
    waiting_for_tools_ms: int = 0,
    # Status block override (default: success)
    status_overrides: dict | None = None,
    # Block 2 — Resources
    cost_reporting_model: str = "token_based",
    token_based: dict | None = None,
    vendor_cost: dict | None = None,
    fixed_subscription: dict | None = None,
    # Optional custom metadata on Status block
    custom_metadata: dict | None = None,
    # Block 4 — Financial overrides
    financial: dict | None = None,
    # Block 5 — Artifacts (list, default empty)
    artifacts: list[dict] | None = None,
    # Block 6 — Quality (default minimal-success)
    quality: dict | None = None,
    # Block 7 — Tool calls (list, default empty)
    tool_calls: list[dict] | None = None,
    # Block 8 — Risk overrides
    risk: dict | None = None,
) -> dict:
    """Build the canonical 8-block payload. All identifier fields are echoed
    in Block 1 from the inbound request. Optional kwargs let the caller fill in
    cost reporting, artifacts, quality, etc.
    """
    total_elapsed = max(0, ended_at_ms - started_at_ms)
    agent_active = max(0, total_elapsed - waiting_for_human_ms - waiting_for_tools_ms)

    # ---- Block 1: Status (with identifier echo per PCD §7) ----
    status = {
        "code": "success",
        "http_status": 200,
        "message": f"{agent_id} completed.",
        "trace_id": request.get("trace_id", UNAVAILABLE),
        "workflow_template_id": request.get("workflow_template_id", UNAVAILABLE),
        "workflow_id": request.get("workflow_id", UNAVAILABLE),
        "agent_run_id": request.get("agent_run_id", UNAVAILABLE),
        "step_id": request.get("step_id", UNAVAILABLE),
        "error_code": None,
        "error_category": "none",
        "retryable": False,
        "custom_metadata": custom_metadata or {},
    }
    if status_overrides:
        status.update(status_overrides)

    # ---- Block 2: Resources ----
    resources_block = _resources_for(
        cost_reporting_model,
        token_based=token_based, vendor_cost=vendor_cost,
        fixed_subscription=fixed_subscription,
    )

    # ---- Block 3: Timing ----
    timing = {
        "total_elapsed_ms": total_elapsed,
        "queue_duration_ms": 0,
        "agent_active_ms": agent_active,
        "waiting_for_tools_ms": waiting_for_tools_ms,
        "waiting_for_human_ms": waiting_for_human_ms,
        "timing_breakdown": UNAVAILABLE,
        "critical_path": UNAVAILABLE,
    }

    # ---- Block 4: Financial (defaults pulled from task complexity) ----
    task_type = request.get("task", {}).get("task_type", UNAVAILABLE)
    complexity = request.get("task", {}).get("complexity_tier", "simple")
    fin = {
        "task_type": task_type,
        "complexity_tier": complexity,
        "manual_baseline_hours": _baseline_hours(task_type, complexity),
        "agent_active_hours": round(agent_active / 3_600_000.0, 6),
        "human_review_hours_estimated": None,
        "agent_cost_usd": 0.0,
        "retry_cost_usd": 0.0,
        "currency": "usd",
    }
    if financial:
        fin.update(financial)

    # ---- Block 6: Quality ----
    q = {
        "confidence": 0.0,
        "initial_attempt_success": True,
        "total_attempts": 1,
        "self_corrections": 0,
        "quality_concerns": [],
        "unsupported_claims": [],
        "completeness_assessment": "all_sections_complete",
    }
    if quality:
        q.update(quality)

    # ---- Block 7: Tool calls ----
    calls = list(tool_calls or [])
    tool_summary = {
        "total_tool_calls": len(calls),
        "total_tool_duration_ms": sum(c.get("duration_ms", 0) for c in calls),
        "tools_as_pct_of_elapsed": (
            int(100 * sum(c.get("duration_ms", 0) for c in calls) / total_elapsed)
            if total_elapsed > 0 else 0
        ),
    }

    # ---- Block 8: Risk ----
    r = {
        "pii_detected": False,
        "pii_filtered_count": 0,
        "secrets_detected": False,
        "policy_violations": 0,
        "sensitivity_compliant": True,
        "missing_inputs": [],
        "unsupported_claims": [],
        "compliance_checks": [],
    }
    if risk:
        r.update(risk)

    return {
        "status": status,
        "resources": resources_block,
        "timing": timing,
        "financial": fin,
        "artifacts": list(artifacts or []),
        "quality": q,
        "tool_calls": {"calls": calls, "tool_summary": tool_summary},
        "risk": r,
    }


def _baseline_hours(task_type: str, complexity: str) -> float | str:
    """Lookup table — manual baseline hours per task_type × complexity."""
    table = {
        ("audio_transcription", "simple"): 0.5,
        ("audio_transcription", "medium"): 1.0,
        ("brd_generation", "simple"): 8.0,
        ("brd_generation", "medium"): 16.0,
        ("brd_generation", "complex"): 32.0,
        ("brd_review", "simple"): 2.0,
        ("brd_review", "medium"): 4.0,
        ("brd_review", "complex"): 8.0,
    }
    return table.get((task_type, complexity), UNAVAILABLE)


def validate_payload(payload: dict) -> None:
    """Enforce PAYLOAD_SCHEMA §9 rules. Raises ValueError on violation."""
    missing = REQUIRED_BLOCKS - set(payload.keys())
    if missing:
        raise ValueError(f"Missing blocks: {sorted(missing)}")
    s = payload["status"]
    for k in ("trace_id", "workflow_template_id", "workflow_id",
               "agent_run_id", "step_id"):
        if not s.get(k):
            raise ValueError(f"status.{k} not echoed")
    if payload["financial"].get("currency") != "usd":
        raise ValueError("financial.currency must be 'usd'")
    if payload["quality"].get("total_attempts", 0) > 3:
        # D8 self-correction cap; orchestrator should have escalated before this
        raise ValueError("total_attempts > 3 (D8 cap)")
```

```python
def _run_transcribe(audio_uri: str, job_name: str) -> dict:
    """Start a transcription job and poll until COMPLETED. Returns the job result."""
    # Detect format from URI extension; webm/opus is treated as ogg by Transcribe.

        resp = get_transcribe().get_transcription_job(TranscriptionJobName=job_name)
        status = resp["TranscriptionJob"]["TranscriptionJobStatus"]
        if status in ("COMPLETED", "FAILED"):
            return resp["TranscriptionJob"]
```

```python
def _read_artifact(uri: str) -> bytes:
    """Read via POSIX mount if available (>1MB content typically claim-checked); fallback to boto3."""
    if uri.startswith("s3://"):
        bucket, key = uri[5:].split("/", 1)
        head = get_s3().head_object(Bucket=bucket, Key=key)
        if head["ContentLength"] >= CLAIM_CHECK_THRESHOLD and os.path.isdir(MOUNT_PATH):
            local = os.path.join(MOUNT_PATH, key)
            with open(local, "rb") as f:
                return f.read()
        return get_s3().get_object(Bucket=bucket, Key=key)["Body"].read()
    raise ValueError(f"Unsupported URI: {uri}")


def _call_claude(prompt: str, max_tokens: int = 4000) -> tuple[str, int, int]:

                         "messages": [{"role": "user", "content": prompt}]}))
    body = json.loads(resp["body"].read())
    text = body["content"][0]["text"]
    return text, body["usage"]["input_tokens"], body["usage"]["output_tokens"]

                reason=f"Self-correction cap ({SELF_CORRECTION_CAP}) exhausted; escalating to human reviewer.")

        transcript = json.loads(transcript_bytes.decode())["results"]["transcripts"][0]["transcript"]

        if _needs_clarification(text):

                               "previous_version_ref": f"s3://{ARTIFACTS_BUCKET}/brd/{request['workflow_id']}/v{attempt-1}.0.md" if attempt > 1 else None,
                               "change_summary": "Initial draft" if attempt == 1 else f"Revised draft (attempt {attempt})"},

def _build_prompt(transcript: str, task: dict, human_input: dict | None) -> str:
    base = (f"You are a Business Analyst. Generate a Business Requirements Document "
            f"(BRD) in Markdown format from the following meeting transcript.\n\n"
            f"Template: {task.get('template_id', 'brd-template-v1')}\n"
            f"Output sections: Overview, Stakeholders, Functional Requirements, "
            f"Non-Functional Requirements, Acceptance Criteria, Open Questions.\n\n"
            f"Transcript:\n{transcript}\n")
    if human_input:
        base += f"\n\nReviewer clarification: {human_input.get('rationale', '')}\n"
    return base


def _needs_clarification(text: str) -> bool:
    # Trivial heuristic for the demo: if the draft contains "[UNCLEAR]" markers or
    # has fewer than 200 chars, ask for clarification.
    return "[UNCLEAR]" in text or len(text) < 200
```

```python
"""Real LangGraph node implementations for Phase 2."""
import re
from typing import TypedDict


class ReviewerState(TypedDict, total=False):
    brd_markdown: str
    quality_findings: list[dict]
    pii_findings: list[dict]
    policy_findings: list[dict]
    final_report: str


def analyze_quality(state: ReviewerState) -> dict:
    """Heuristic quality check: section completeness, sentence count, vagueness markers."""
    md = state["brd_markdown"]
    findings = []
    required_sections = ["Overview", "Stakeholders", "Functional Requirements",
                         "Non-Functional Requirements", "Acceptance Criteria"]
    for sec in required_sections:
        if sec.lower() not in md.lower():
            findings.append({"severity": "medium", "summary": f"Missing section: {sec}"})
    if md.count("TBD") > 0 or md.count("[UNCLEAR]") > 0:
        findings.append({"severity": "low",
                         "summary": f"Contains {md.count('TBD') + md.count('[UNCLEAR]')} placeholder marker(s)"})
    return {"quality_findings": findings}


PII_PATTERNS = [
    (re.compile(r"\b\d{3}-\d{2}-\d{4}\b"), "ssn"),
    (re.compile(r"\b\d{4}[-\s]?\d{4}[-\s]?\d{4}[-\s]?\d{4}\b"), "credit_card"),
    (re.compile(r"\b[\w.+-]+@[\w-]+\.[\w.-]+\b"), "email"),
    (re.compile(r"\b(?:\+?\d{1,3}[-.\s]?)?\(?\d{3}\)?[-.\s]?\d{3}[-.\s]?\d{4}\b"), "phone"),
]


def scan_pii(state: ReviewerState) -> dict:
    """Regex-based PII detection. Production would also call Comprehend DetectPiiEntities."""
    md = state["brd_markdown"]
    findings = []
    for pattern, kind in PII_PATTERNS:
        for match in pattern.findall(md):
            findings.append({"type": kind, "summary": f"{kind} pattern detected"})
    return {"pii_findings": findings}


def check_policy(state: ReviewerState) -> dict:
    """Demo policy rules: forbid certain disallowed phrases."""
    md = state["brd_markdown"].lower()
    findings = []
    for forbidden in ["confidential do not share", "internal use only - do not store"]:
        if forbidden in md:
            findings.append({"summary": f"Forbidden phrase: '{forbidden}'"})
    return {"policy_findings": findings}


def generate_report(state: ReviewerState) -> dict:
    """Compile findings into a Markdown review report."""
    lines = ["# BRD Review Report\n"]
    lines.append(f"## Quality Findings ({len(state.get('quality_findings', []))})\n")
    for f in state.get("quality_findings", []):
        lines.append(f"- [{f.get('severity', '?')}] {f['summary']}")
    lines.append(f"\n## PII Findings ({len(state.get('pii_findings', []))})\n")
    for f in state.get("pii_findings", []):
        lines.append(f"- {f['type']}: {f['summary']}")
    lines.append(f"\n## Policy Findings ({len(state.get('policy_findings', []))})\n")
    for f in state.get("policy_findings", []):
        lines.append(f"- {f['summary']}")
    return {"final_report": "\n".join(lines) + "\n"}
```
